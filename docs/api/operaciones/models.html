 37class TasaDeCambio(models.Model):
 38    """Tasa de cambio de una moneda respecto al guaraní (PYG).
 39
 40    Cada registro es histórico: la tasa vigente es la más reciente
 41    (ordenadas por `fecha_vigencia` descendente).
 42
 43    Attributes:
 44        moneda: Moneda a la que pertenece la tasa.
 45        tasa_base: Valor de referencia en PYG.
 46        margen_compra: Monto que se resta a la base para el precio de compra.
 47        margen_venta: Monto que se suma a la base para el precio de venta.
 48        fecha_vigencia: Fecha y hora de registro de la tasa.
 49    """
 50    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
 51    moneda = models.ForeignKey(Moneda, on_delete=models.CASCADE, related_name='tasas')
 52    tasa_base = models.DecimalField(max_digits=12, decimal_places=4)
 53    margen_compra = models.DecimalField(max_digits=12, decimal_places=4)
 54    margen_venta = models.DecimalField(max_digits=12, decimal_places=4)
 55    fecha_vigencia = models.DateTimeField(auto_now_add=True)
 56
 57    class Meta:
 58        """Nombres legibles y orden de más reciente a más antigua."""
 59        verbose_name = "Tasa de Cambio"
 60        verbose_name_plural = "Tasas de Cambio"
 61        ordering = ['-fecha_vigencia']
 62
 63    def __str__(self):
 64        """Devuelve `MONEDA @ dd/mm/aaaa hh:mm`."""
 65        return f"{self.moneda.codigo} @ {self.fecha_vigencia:%d/%m/%Y %H:%M}"
 66
 67    def calcular_precio_compra(self):
 68        """Calcula el precio al que la casa compra la moneda.
 69
 70        Returns:
 71            Decimal: `tasa_base - margen_compra`.
 72        """
 73        return self.tasa_base - self.margen_compra
 74
 75    def calcular_precio_venta(self):
 76        """Calcula el precio al que la casa vende la moneda.
 77
 78        Returns:
 79            Decimal: `tasa_base + margen_venta`.
 80        """
 81        return self.tasa_base + self.margen_venta

 84class Transaccion(models.Model):
 85    """Operación de compra o venta de divisas realizada en nombre de un cliente.
 86
 87    Un cliente puede tener varios usuarios asociados que operan en su nombre
 88    y un usuario puede operar para varios clientes: `cliente` indica a nombre
 89    de quién se hizo la operación y `usuario`, quién la realizó.
 90
 91    El tipo se interpreta desde el punto de vista del cliente:
 92
 93    - `COMPRA`: el cliente paga guaraníes (PYG) y recibe la divisa.
 94    - `VENTA`: el cliente paga la divisa y recibe guaraníes (PYG).
 95
 96    Los registros son de auditoría: las claves foráneas usan `PROTECT` para
 97    que el historial no se pierda si se intenta borrar el cliente, el usuario,
 98    una moneda o un medio de pago.
 99
100    Attributes:
101        id: Identificador UUID, no editable.
102        cliente: Cliente en nombre del cual se realizó la operación.
103        usuario: Usuario asociado al cliente que realizó la operación.
104        cajero: Usuario (rol `Cajero`) que confirmó la operación. Vacío
105            mientras la transacción esté `PENDIENTE`.
106        tipo: `COMPRA` o `VENTA`.
107        moneda: Divisa operada.
108        medio_pago: Medio de pago utilizado.
109        monto_pagado: Monto entregado por el cliente.
110        monto_recibido: Monto recibido por el cliente.
111        tasa_aplicada: Precio en PYG de una unidad de la divisa.
112        facturada: Si se emitió factura de la operación.
113        dispositivo: Tipo de dispositivo desde el que se realizó
114            (ver `operaciones.utils.detectar_dispositivo`).
115        estado: `PENDIENTE`, `CONFIRMADA` o `CANCELADA`.
116        fecha: Fecha y hora de la operación.
117    """
118
119    class Tipo(models.TextChoices):
120        """Tipo de operación desde el punto de vista del cliente."""
121        COMPRA = 'COMPRA', 'Compra'
122        VENTA = 'VENTA', 'Venta'
123
124    class Dispositivo(models.TextChoices):
125        """Dispositivos reconocidos a partir del User-Agent del navegador."""
126        IPHONE = 'IPHONE', 'iPhone'
127        IPAD = 'IPAD', 'iPad'
128        ANDROID = 'ANDROID', 'Android'
129        WINDOWS = 'WINDOWS', 'PC (Windows)'
130        MAC = 'MAC', 'Mac'
131        LINUX = 'LINUX', 'PC (Linux)'
132        OTRO = 'OTRO', 'Otro'
133
134    class Estado(models.TextChoices):
135        """Estados de la transacción (la lógica de cambio de estado es de otra historia)."""
136        PENDIENTE = 'PENDIENTE', 'Pendiente'
137        CONFIRMADA = 'CONFIRMADA', 'Confirmada'
138        CANCELADA = 'CANCELADA', 'Cancelada'
139
140    MONEDA_LOCAL = 'PYG'
141
142    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
143    cliente = models.ForeignKey('clientes.Cliente', on_delete=models.PROTECT, related_name='transacciones')
144    usuario = models.ForeignKey(
145        settings.AUTH_USER_MODEL, on_delete=models.PROTECT, related_name='transacciones_realizadas'
146    )
147    cajero = models.ForeignKey(
148        settings.AUTH_USER_MODEL, on_delete=models.SET_NULL, null=True, blank=True,
149        related_name='transacciones_atendidas'
150    )
151    tipo = models.CharField(max_length=10, choices=Tipo.choices)
152    moneda = models.ForeignKey(Moneda, on_delete=models.PROTECT, related_name='transacciones')
153    medio_pago = models.ForeignKey('mpagos.MedioPago', on_delete=models.PROTECT, related_name='transacciones')
154    monto_pagado = models.DecimalField(max_digits=18, decimal_places=2)
155    monto_recibido = models.DecimalField(max_digits=18, decimal_places=2)
156    tasa_aplicada = models.DecimalField(max_digits=12, decimal_places=4)
157    facturada = models.BooleanField(default=False)
158    dispositivo = models.CharField(max_length=10, choices=Dispositivo.choices, default=Dispositivo.OTRO)
159    estado = models.CharField(max_length=10, choices=Estado.choices, default=Estado.PENDIENTE)
160    # default en lugar de auto_now_add para poder registrar operaciones con fecha pasada
161    fecha = models.DateTimeField(default=timezone.now)
162
163    class Meta:
164        """Nombres legibles, orden de más reciente a más antigua e índice por cliente."""
165        verbose_name = "Transacción"
166        verbose_name_plural = "Transacciones"
167        ordering = ['-fecha']
168        indexes = [models.Index(fields=['cliente', '-fecha'])]
169
170    def __str__(self):
171        """Devuelve `TIPO MONEDA - cliente (dd/mm/aaaa hh:mm)`."""
172        return f"{self.get_tipo_display()} {self.moneda.codigo} - {self.cliente} ({self.fecha:%d/%m/%Y %H:%M})"
173
174    @property
175    def moneda_pagada(self):
176        """Código de la moneda que entregó el cliente.
177
178        Returns:
179            str: `PYG` en una compra; el código de la divisa en una venta.
180        """
181        return self.MONEDA_LOCAL if self.tipo == self.Tipo.COMPRA else self.moneda.codigo
182
183    @property
184    def moneda_recibida(self):
185        """Código de la moneda que recibió el cliente.
186
187        Returns:
188            str: El código de la divisa en una compra; `PYG` en una venta.
189        """
190        return self.moneda.codigo if self.tipo == self.Tipo.COMPRA else self.MONEDA_LOCAL

Operación de compra o venta de divisas realizada en nombre de un cliente.

Un cliente puede tener varios usuarios asociados que operan en su nombre y un usuario puede operar para varios clientes: cliente indica a nombre de quién se hizo la operación y usuario, quién la realizó.

El tipo se interpreta desde el punto de vista del cliente:

  • COMPRA: el cliente paga guaraníes (PYG) y recibe la divisa.
  • VENTA: el cliente paga la divisa y recibe guaraníes (PYG).

Los registros son de auditoría: las claves foráneas usan PROTECT para que el historial no se pierda si se intenta borrar el cliente, el usuario, una moneda o un medio de pago.

Atributos:
  • id: Identificador UUID, no editable.
  • cliente: Cliente en nombre del cual se realizó la operación.
  • usuario: Usuario asociado al cliente que realizó la operación.
  • cajero: Usuario (rol Cajero) que confirmó la operación. Vacío mientras la transacción esté PENDIENTE.
  • tipo: COMPRA o VENTA.
  • moneda: Divisa operada.
  • medio_pago: Medio de pago utilizado.
  • monto_pagado: Monto entregado por el cliente.
  • monto_recibido: Monto recibido por el cliente.
  • tasa_aplicada: Precio en PYG de una unidad de la divisa.
  • facturada: Si se emitió factura de la operación.
  • dispositivo: Tipo de dispositivo desde el que se realizó (ver operaciones.utils.detectar_dispositivo).
  • estado: PENDIENTE, CONFIRMADA o CANCELADA.
  • fecha: Fecha y hora de la operación.
