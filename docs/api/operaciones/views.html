147    - genera una notificación para los usuarios del sistema.

154

157            moneda = form.cleaned_data['moneda']
158
159            tasa_anterior = TasaDeCambio.objects.filter(moneda=moneda).order_by('-fecha_vigencia').first()
160
161            with transaction.atomic():
162
163                tasa_nueva = form.save()
164
165                crear_notificacion_cambio_tasa(
166                    tasa_nueva=tasa_nueva,
167                    tasa_anterior=tasa_anterior,
168                    usuario=request.user
169                )
170
171            messages.success(
172                request,
173                f"La tasa de {moneda.codigo} fue actualizada correctamente."
174            )
175
176            return redirect('lista_tasas')
177
178    else:
179        form = TasaDeCambioForm(initial={'moneda': moneda_inicial} if moneda_inicial else None)

183def simular(request):
184    """Simulador de conversión de una moneda extranjera a guaraníes (PYG).
185
186    Lee de la query string `monto`, `tasa_id` y `tipo_operacion`
187    (`compra` o `venta`) y calcula el monto resultante con el precio
188    de compra o de venta de la tasa elegida. Es de acceso público.
189
190    Args:
191        request: Petición HTTP (GET).
192
193    Returns:
194        HttpResponse: Plantilla `operaciones/simulador.html` con el resultado
195        o un mensaje de error si los datos no son válidos.
196    """
197
198    resultado = None
199    monto_ingresado = request.GET.get('monto', '')
200    tasa_id = request.GET.get('tasa_id', '')
201    tipo_op = request.GET.get('tipo_operacion', 'compra')
202
203    #Obtener las divisas
204    tasas = TasaDeCambio.objects.select_related('moneda').filter(
205        moneda__habilitada=True
206    ).order_by('-fecha_vigencia')
207
208    if monto_ingresado and tasa_id:
209        try:
210            monto = Decimal(monto_ingresado)
211            tasa = TasaDeCambio.objects.get(id=tasa_id)
212            
213            # Ejecución de los métodos definidos en la entidad TasaDeCambio
214            if tipo_op == 'compra':
215                precio_aplicado = tasa.calcular_precio_compra()
216            else:
217                precio_aplicado = tasa.calcular_precio_venta()
218
219            monto_destino = monto * precio_aplicado
220
221            resultado = {
222                'monto_origen': monto,
223                'monto_destino': monto_destino,
224                'precio_aplicado': precio_aplicado,
225                'moneda_origen': tasa.moneda.codigo,
226                'moneda_destino': 'PYG',
227                'tipo_operacion': tipo_op,
228            }
229        except (InvalidOperation, TasaDeCambio.DoesNotExist):
230            resultado = {'error': 'Por favor ingrese un monto y una tasa de cambio válidos.'}
231
232    context = {
233        'tasas': tasas,
234        'resultado': resultado,
235        'monto_input': monto_ingresado,
236        'tasa_id_sel': tasa_id,
237        'tipo_op_sel': tipo_op,
238    }
239    return render(request, 'operaciones/simulador.html', context)

241def obtener_cliente_activo(request):
242    """
243    Devuelve el cliente activo solo si pertenece
244    realmente al usuario autenticado.
245    """
246    clientes = list(
247        Cliente.objects.filter(
248            usuarios_asociados__usuario=request.user
249        )
250    )
251
252    if not clientes:
253        return None
254
255    cliente_id = request.session.get(
256        'ge_active_client'
257    )
258
259    cliente = next(
260        (
261            c for c in clientes
262            if str(c.id) == str(cliente_id)
263        ),
264        clientes[0])
265    request.session['ge_active_client'] = str(cliente.id)
266
267    return cliente

275@require_GET
276def historial_transacciones(request):
277    """Lista, en modo solo lectura, las transacciones del cliente activo.
278
279    Acceso: usuario autenticado con al menos un `Cliente` asociado. Si no lo
280    tiene, se lo redirige a `convertirse_en_cliente`. Un usuario anónimo es
281    redirigido al login.
282
283    Se muestran todas las transacciones realizadas en nombre del cliente
284    activo (ver `_cliente_activo`), incluidas las de otros usuarios asociados
285    a ese cliente, indicando quién operó cada una. Nunca se exponen
286    operaciones de clientes que no estén asociados al usuario. La vista solo
287    acepta GET: no crea, modifica ni elimina transacciones.
288
289    Los filtros llegan por query string (ver `FiltroHistorialForm`) y el
290    resultado se pagina de a `TRANSACCIONES_POR_PAGINA`, de la más reciente
291    a la más antigua. `rango_paginas` contiene los números de página a
292    mostrar, abreviados con `Paginator.ELLIPSIS` cuando hay muchas.
293
294    Args:
295        request: Petición HTTP (GET con filtros opcionales y `page`).
296
297    Returns:
298        HttpResponse: Plantilla `operaciones/historial.html`, o redirección a
299        `convertirse_en_cliente`.
300    """
301    cliente = obtener_cliente_activo(request)
302    if cliente is None:
303        return redirect('convertirse_en_cliente')
304
305    # Límite de seguridad: solo transacciones del cliente activo (asociado al usuario)
306    transacciones = Transaccion.objects.filter(cliente=cliente).select_related(
307        'usuario', 'moneda', 'medio_pago', 'cajero'
308    )
309
310    form = FiltroHistorialForm(request.GET or None, transacciones=transacciones)
311    if form.is_bound:
312        form.is_valid()  # completa cleaned_data; los campos con error se ignoran al filtrar
313    filtradas = form.filtrar(transacciones).order_by('-fecha')
314
315    page_obj = Paginator(filtradas, TRANSACCIONES_POR_PAGINA).get_page(request.GET.get('page'))
316
317    context = {
318        'cliente': cliente,
319        'form': form,
320        'page_obj': page_obj,
321        'rango_paginas': list(page_obj.paginator.get_elided_page_range(
322            page_obj.number, on_each_side=PAGINAS_A_CADA_LADO, on_ends=PAGINAS_EN_EXTREMOS
323        )),
324        'total': page_obj.paginator.count,
325        'hay_filtros': any(v for k, v in request.GET.items() if k != 'page'),
326        'menu_sections': get_menu_sections(
327            resolve_user_role(request), cliente, is_authenticated=True, roles=resolve_user_roles(request)
328        ),
329    }
330    return render(request, 'operaciones/historial.html', context)

332@login_required
333def operar(request):
334
335    cliente = obtener_cliente_activo(request)
336
337    if not cliente:
338        messages.warning(
339            request,
340            "Debe tener un cliente asociado y seleccionado "
341            "para realizar operaciones."
342        )
343
344        return redirect('dashboard')
345
346    obtener_o_crear_medio_efectivo(
347        cliente=cliente,
348        usuario=request.user
349    )
350
351    simulacion = None
352
353    if request.method == 'POST':
354
355        form = OperacionForm(
356            request.POST,
357            cliente=cliente
358        )
359
360        if form.is_valid():
361
362            accion = request.POST.get('accion')
363
364            tipo_operacion = form.cleaned_data[
365                'tipo_operacion'
366            ]
367
368            moneda = form.cleaned_data['moneda']
369            monto = form.cleaned_data['monto']
370
371            medio_pago = form.cleaned_data[
372                'medio_pago'
373            ]
374
375            if accion == 'simular':
376
377                try:
378                    simulacion = simular_operacion(
379                        cliente=cliente,
380                        tipo_operacion=tipo_operacion,
381                        moneda=moneda,
382                        monto=monto,
383                    )
384
385                except ValidationError as error:
386                    form.add_error(
387                        None,
388                        error.message
389                    )
390
391            elif accion == 'confirmar':
392                tasa_id = request.POST.get('tasa_id_simulada')
393
394                if not tasa_id:
395                    form.add_error(None, "Debe simular la operación antes de confirmarla.")
396                else:
397                    try:
398                        # 1. DELAY DE 20 SEGUNDOS (Da tiempo a cambiar la tasa en la otra pestaña)
399                        import time
400                        time.sleep(10)
401
402                        # 2. INTENTA CREAR LA TRANSACCIÓN
403                        transaccion = crear_transaccion(
404                            usuario=request.user,
405                            cliente=cliente,
406                            tipo_operacion=tipo_operacion,
407                            moneda=moneda,
408                            monto=monto,
409                            medio_pago=medio_pago,
410                            tasa_id_simulada=tasa_id,
411                        )
412
413                        messages.success(request, "La operación fue registrada correctamente.")
414                        return redirect('detalle_transaccion', transaccion_id=transaccion.id)
415
416                    except CotizacionDesactualizada as error:
417                        # 3. SI EN LOS 20 SEGUNDOS CAMBIO LA TASA, ENTRA AQUÍ
418                        transaccion_cancelada = registrar_transaccion_cancelada(
419                            usuario=request.user,
420                            cliente=cliente,
421                            tipo_operacion=tipo_operacion,
422                            moneda=moneda,
423                            monto=monto,
424                            medio_pago=medio_pago,
425                        )
426
427                        context_cancelado = {
428                            'cliente': cliente,
429                            'transaccion': transaccion_cancelada,
430                            'motivo': str(error),
431                            'moneda': moneda,
432                        }
433                        return render(request, 'operaciones/operacion_cancelada.html', context_cancelado)
434
435                    except ValidationError as error:
436                        form.add_error(None, error.message if hasattr(error, 'message') else str(error))
437       
438
439    else:
440
441        form = OperacionForm(
442            cliente=cliente
443        )
444
445    context = {
446        'form': form,
447        'cliente': cliente,
448        'simulacion': simulacion,
449    }
450
451    return render(
452        request,
453        'operaciones/operar.html',
454        context
455    )

586@csrf_exempt
587@require_POST
588def stripe_webhook(request):
589    """
590    Recibe eventos firmados de Stripe y confirma
591    las operaciones pagadas.
592    """
593
594    payload = request.body
595
596    signature = request.META.get(
597        'HTTP_STRIPE_SIGNATURE'
598    )
599
600    if not settings.STRIPE_WEBHOOK_SECRET:
601        return HttpResponse(status=500)
602
603    try:
604        event = stripe.Webhook.construct_event(
605            payload,
606            signature,
607            settings.STRIPE_WEBHOOK_SECRET,
608        )
609
610    except Exception as error:
611        print(
612            "ERROR FIRMA WEBHOOK STRIPE:",
613            type(error).__name__,
614            repr(error),
615            flush=True
616        )
617
618        return HttpResponse(status=400)
619
620    if event.type != 'payment_intent.succeeded':
621        return HttpResponse(status=200)
622
623    payment_intent = event.data.object
624
625    metadata = (
626        payment_intent.metadata.to_dict()
627        if payment_intent.metadata
628        else {}
629    )
630
631    transaccion_id = metadata.get(
632        'transaccion_id'
633    )
634
635    if not transaccion_id:
636        return HttpResponse(status=200)
637
638    try:
639        transaccion = (
640            Transaccion.objects
641            .select_related(
642                'cliente',
643                'medio_pago',
644                'usuario'
645            )
646            .get(id=transaccion_id)
647        )
648
649    except Transaccion.DoesNotExist:
650        return HttpResponse(status=404)
651
652    # Comprobar que Stripe procesó el Customer
653    # correspondiente al cliente de la operación.
654    if (
655        payment_intent.customer
656        != transaccion.cliente.stripe_customer_id
657    ):
658        return HttpResponse(status=400)
659
660    # Comprobar la tarjeta utilizada.
661    if (
662        payment_intent.payment_method
663        != transaccion.medio_pago.stripe_payment_method_id
664    ):
665        return HttpResponse(status=400)
666
667    transaccion.refresh_from_db()
668
669    # Stripe puede reenviar el mismo webhook.
670    # Si ya está pagada, respondemos 200 sin duplicar el log.
671    if (
672        transaccion.estado
673        == Transaccion.Estado.PAGADA
674    ):
675        return HttpResponse(status=200)
676
677    if (
678        transaccion.estado
679        != Transaccion.Estado.PENDIENTE
680    ):
681        return HttpResponse(status=200)
682
683    cambiar_estado_transaccion(
684        transaccion=transaccion,
685        nuevo_estado=Transaccion.Estado.PAGADA,
686        usuario=transaccion.usuario,
687        motivo=(
688            "Pago confirmado mediante "
689            "Stripe Sandbox."
690        ),
691    )
692
693    print(
694        "STRIPE: Transacción",
695        transaccion.id,
696        "confirmada como PAGADA.",
697        flush=True
698    )
699
700    return HttpResponse(status=200)

587@csrf_exempt
588@require_POST
589def stripe_webhook(request):
590    """
591    Recibe eventos firmados de Stripe y confirma
592    las operaciones pagadas.
593    """
594
595    payload = request.body
596
597    signature = request.META.get(
598        'HTTP_STRIPE_SIGNATURE'
599    )
600
601    if not settings.STRIPE_WEBHOOK_SECRET:
602        return HttpResponse(status=500)
603
604    try:
605        event = stripe.Webhook.construct_event(
606            payload,
607            signature,
608            settings.STRIPE_WEBHOOK_SECRET,
609        )
610
611    except Exception as error:
612        print(
613            "ERROR FIRMA WEBHOOK STRIPE:",
614            type(error).__name__,
615            repr(error),
616            flush=True
617        )
618
619        return HttpResponse(status=400)
620
621    if event.type != 'payment_intent.succeeded':
622        return HttpResponse(status=200)
623
624    payment_intent = event.data.object
625
626    metadata = (
627        payment_intent.metadata.to_dict()
628        if payment_intent.metadata
629        else {}
630    )
631
632    transaccion_id = metadata.get(
633        'transaccion_id'
634    )
635
636    if not transaccion_id:
637        return HttpResponse(status=200)
638
639    try:
640        transaccion = (
641            Transaccion.objects
642            .select_related(
643                'cliente',
644                'medio_pago',
645                'usuario'
646            )
647            .get(id=transaccion_id)
648        )
649
650    except Transaccion.DoesNotExist:
651        return HttpResponse(status=404)
652
653    # Comprobar que Stripe procesó el Customer
654    # correspondiente al cliente de la operación.
655    if (
656        payment_intent.customer
657        != transaccion.cliente.stripe_customer_id
658    ):
659        return HttpResponse(status=400)
660
661    # Comprobar la tarjeta utilizada.
662    if (
663        payment_intent.payment_method
664        != transaccion.medio_pago.stripe_payment_method_id
665    ):
666        return HttpResponse(status=400)
667
668    transaccion.refresh_from_db()
669
670    # Stripe puede reenviar el mismo webhook.
671    # Si ya está pagada, respondemos 200 sin duplicar el log.
672    if (
673        transaccion.estado
674        == Transaccion.Estado.PAGADA
675    ):
676        return HttpResponse(status=200)
677
678    if (
679        transaccion.estado
680        != Transaccion.Estado.PENDIENTE
681    ):
682        return HttpResponse(status=200)
683
684    cambiar_estado_transaccion(
685        transaccion=transaccion,
686        nuevo_estado=Transaccion.Estado.PAGADA,
687        usuario=transaccion.usuario,
688        motivo=(
689            "Pago confirmado mediante "
690            "Stripe Sandbox."
691        ),
692    )
693
694    print(
695        "STRIPE: Transacción",
696        transaccion.id,
697        "confirmada como PAGADA.",
698        flush=True
699    )
700
701    return HttpResponse(status=200)
