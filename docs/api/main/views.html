 24def resolve_keycloak_role(user_roles):
 25    """Devuelve la etiqueta interna del rol principal según los roles reales de Keycloak."""
 26    if not user_roles:

245def select_client(request):
246    """Endpoint AJAX para cambiar el cliente activo del usuario.
247
248    Espera un POST con JSON `{"client_id": "<uuid>"}`. Solo permite
249    seleccionar clientes asociados al usuario.
250
251    Args:
252        request: Petición HTTP (POST con cuerpo JSON).
253
254    Returns:
255        JsonResponse: `{"status": "ok"}`, 403 si el cliente no le pertenece
256        o 400 si el método no es POST.
257    """
258    if request.method == "POST":
259        data = json.loads(request.body)
260        client_id = data.get("client_id")
261        if Cliente.objects.filter(id=client_id, usuarios_asociados__usuario=request.user).exists():
262            request.session['ge_active_client'] = client_id
263            return JsonResponse({"status": "ok"})
264        return JsonResponse({"status": "forbidden", "message": "No tiene permisos sobre este cliente"}, status=403)
265    return JsonResponse({"status": "error"}, status=400)

265        return JsonResponse({"status": "forbidden", "message": "No tiene permisos sobre este cliente"}, status=403)
266    return JsonResponse({"status": "error"}, status=400)
