 11class ArqueoCaja(models.Model):
 12    """Registro formal del cierre de una caja.
 13
 14    Guarda una "foto" de la caja al momento del cierre (identificador, nombre
 15    y sucursal) en lugar de una clave foránea al modelo de caja, que pertenece
 16    a otra historia. Así el arqueo no depende de ese modelo y conserva los
 17    datos aunque la caja se renombre o se elimine.
 18
 19    Es un registro de auditoría: una vez guardado no se puede modificar
 20    (ver `save`). El detalle por moneda está en `DetalleArqueo`.
 21
 22    Attributes:
 23        id: Identificador UUID, no editable.
 24        caja_id: Identificador de la caja cerrada (clave primaria en su modelo).
 25        caja_nombre: Nombre de la caja al momento del cierre.
 26        sucursal_nombre: Sucursal de la caja al momento del cierre.
 27        cajero: Usuario que realizó el cierre.
 28        fecha_cierre: Fecha y hora del cierre.
 29    """
 30    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
 31    caja_id = models.CharField(max_length=64)
 32    caja_nombre = models.CharField(max_length=150)
 33    sucursal_nombre = models.CharField(max_length=150, blank=True)
 34    cajero = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.PROTECT, related_name='arqueos')
 35    fecha_cierre = models.DateTimeField(default=timezone.now)
 36
 37    class Meta:
 38        """Nombres legibles y orden del más reciente al más antiguo."""
 39        verbose_name = "Arqueo de caja"
 40        verbose_name_plural = "Arqueos de caja"
 41        ordering = ['-fecha_cierre']
 42
 43    def __str__(self):
 44        """Devuelve `Arqueo <caja> - dd/mm/aaaa hh:mm`."""
 45        return f"Arqueo {self.caja_nombre} - {self.fecha_cierre:%d/%m/%Y %H:%M}"
 46
 47    def save(self, *args, **kwargs):
 48        """Guarda el arqueo solo la primera vez.
 49
 50        Raises:
 51            ValidationError: Si se intenta modificar un arqueo ya registrado.
 52        """
 53        if not self._state.adding:
 54            raise ValidationError("Un arqueo registrado no se puede modificar.")
 55        super().save(*args, **kwargs)
 56
 57    @property
 58    def tiene_discrepancias(self):
 59        """Indica si alguna moneda tuvo sobrante o faltante.
 60
 61        Returns:
 62            bool: `True` si al menos un detalle no cuadra.
 63        """
 64        return self.detalles.exclude(resultado=DetalleArqueo.Resultado.CUADRA).exists()

 67class DetalleArqueo(models.Model):
 68    """Resultado del arqueo para una moneda.
 69
 70    Attributes:
 71        arqueo: Arqueo al que pertenece.
 72        moneda: Código ISO 4217 de la moneda (ej. `PYG`, `USD`).
 73        saldo_registrado: Saldo que el sistema esperaba en la caja.
 74        saldo_declarado: Saldo físico contado y declarado por el cajero.
 75        diferencia: `saldo_declarado - saldo_registrado`.
 76        resultado: `CUADRA`, `SOBRANTE` o `FALTANTE`.
 77    """
 78
 79    class Resultado(models.TextChoices):
 80        """Resultado de comparar el saldo declarado con el registrado."""
 81        CUADRA = 'CUADRA', 'Cuadra'
 82        SOBRANTE = 'SOBRANTE', 'Sobrante'
 83        FALTANTE = 'FALTANTE', 'Faltante'
 84
 85    arqueo = models.ForeignKey(ArqueoCaja, on_delete=models.CASCADE, related_name='detalles')
 86    moneda = models.CharField(max_length=3)
 87    saldo_registrado = models.DecimalField(max_digits=18, decimal_places=2)
 88    saldo_declarado = models.DecimalField(max_digits=18, decimal_places=2)
 89    diferencia = models.DecimalField(max_digits=18, decimal_places=2)
 90    resultado = models.CharField(max_length=10, choices=Resultado.choices)
 91
 92    class Meta:
 93        """Una fila por moneda en cada arqueo, ordenadas por código."""
 94        verbose_name = "Detalle de arqueo"
 95        verbose_name_plural = "Detalles de arqueo"
 96        ordering = ['moneda']
 97        constraints = [
 98            models.UniqueConstraint(fields=['arqueo', 'moneda'], name='detalle_arqueo_moneda_unica'),
 99        ]
100
101    def __str__(self):
102        """Devuelve `MONEDA: Resultado (diferencia)`."""
103        return f"{self.moneda}: {self.get_resultado_display()} ({self.diferencia})"
104
105    def save(self, *args, **kwargs):
106        """Guarda el detalle solo la primera vez.
107
108        Raises:
109            ValidationError: Si se intenta modificar un detalle ya registrado.
110        """
111        if not self._state.adding:
112            raise ValidationError("Un detalle de arqueo registrado no se puede modificar.")
113        super().save(*args, **kwargs)

Registro formal del cierre de una caja.

Guarda una "foto" de la caja al momento del cierre (identificador, nombre y sucursal) en lugar de una clave foránea al modelo de caja, que pertenece a otra historia. Así el arqueo no depende de ese modelo y conserva los datos aunque la caja se renombre o se elimine.

Es un registro de auditoría: una vez guardado no se puede modificar (ver save). El detalle por moneda está en DetalleArqueo.

Atributos:
  • id: Identificador UUID, no editable.
  • caja_id: Identificador de la caja cerrada (clave primaria en su modelo).
  • caja_nombre: Nombre de la caja al momento del cierre.
  • sucursal_nombre: Sucursal de la caja al momento del cierre.
  • cajero: Usuario que realizó el cierre.
  • fecha_cierre: Fecha y hora del cierre.
