mpagos.views

Módulo de controladores/vistas para la gestión de Medios de Pago (mpagos).

Implementa la lógica del CRUD (Crear, Leer, Modificar, Eliminar) asegurando que cada cliente acceda e interactúe únicamente con sus propios medios de pago.

 24@login_required
 25def listar_medios_pago(request):
 26    """
 27    Vista para consultar y listar todos los medios de pago activos del usuario autenticado.
 28    """
 29    cliente = obtener_cliente_activo(request)
 30
 31    if not cliente:

 33        return redirect('dashboard')
 34    
 35    medios = (MedioPago.objects.filter(cliente=cliente, activo=True).exclude(tipo='EFECTIVO'))
 36    puede_administrar = puede_administrar_cliente(request, cliente)
 37    return render(request, 'mpagos/listar.html', {'medios': medios, 'cliente' : cliente, 'puede_administrar' : puede_administrar})

 40@login_required
 41def crear_medio_pago(request):
 42    """
 43    Registra un medio de pago para el cliente activo.
 44
 45    La operación solo puede ser realizada por un Administrador General
 46    o por un usuario con rol local ADMIN en el cliente seleccionado.
 47
 48    El medio queda asociado al cliente activo y se registra también
 49    el usuario que realizó la creación.
 50
 51    Args:
 52        request:
 53            Solicitud HTTP de Django.
 54
 55    Returns:
 56        HttpResponse:
 57            Formulario de creación cuando la petición es GET o contiene
 58            errores de validación.
 59
 60        HttpResponseRedirect:
 61            Redirección al listado después de crear correctamente
 62            el medio de pago.
 63
 64        HttpResponseForbidden:
 65            Respuesta 403 si el usuario no puede administrar el cliente.
 66    """
 67
 68    cliente = obtener_cliente_activo(request)
 69
 70    if not cliente:
 71        messages.warning(request, "Debe seleccionar un cliente.")
 72        return redirect('dashboard')
 73
 74    if not puede_administrar_cliente(request, cliente):
 75        return HttpResponseForbidden("No tiene permisos para agregar medios de pago a este cliente.")
 76
 77    if request.method == 'POST':
 78
 79        # Las tarjetas se registrane exclusivamente mediante Stripe Elements
 80
 81        if request.POST.get('tipo') == 'TARJETA':
 82             
 83            form = MedioPagoForm(request.POST)
 84
 85            form.add_error(None, "Las tarjetas deben registrarse mediante el formulario seguro")
 86
 87        else:
 88
 89            form = MedioPagoForm(request.POST)
 90
 91            if form.is_valid():
 92                medio = form.save(commit=False)
 93                medio.usuario = request.user
 94                medio.cliente = cliente
 95
 96            if medio.es_predeterminado:
 97                MedioPago.objects.filter(cliente=cliente, activo = True).update(es_predeterminado=False)
 98            medio.save()
 99            messages.success(request, "Medio de pago registrado exitosamente.")
100            return redirect('mpagos:listar')
101    else:
102        form = MedioPagoForm()
103
104    return render(request, 'mpagos/form.html', {
105        'form': form,
106        'cliente' : cliente,
107        'titulo': 'Registrar Medio de Pago',
108        'stripe_publishable_key': (settings.STRIPE_PUBLISHABLE_KEY),
109    })

112@login_required
113def editar_medio_pago(request, pk):
114    """
115    Vista para modificar los datos de un medio de pago existente perteneciente al cliente activo.
116    """
117    cliente = obtener_cliente_activo(request)
118
119    if not cliente:
120        messages.warning(request,"Debe seleccionar un cliente.")
121        return redirect('dashboard')
122
123    if not puede_administrar_cliente(request, cliente):
124        return HttpResponseForbidden(
125            "No tiene permisos para editar los medios de pago de este cliente.")
126
127    medio = get_object_or_404(MedioPago, pk=pk, cliente=cliente, activo=True, tipo__in=['TARJETA','SIPAP'])
128
129    if request.method == 'POST':
130        form = MedioPagoForm(request.POST, instance=medio)
131        if form.is_valid():
132            medio_editado = form.save(commit=False)
133
134            medio_editado.cliente = cliente
135
136            if medio_editado.es_predeterminado:
137                MedioPago.objects.filter(cliente=cliente, activo=True).exclude(pk=pk).update(es_predeterminado=False)
138
139            medio_editado.save()
140            messages.success(request, "Medio de pago actualizado correctamente.")
141            return redirect('mpagos:listar')
142    else:
143        form = MedioPagoForm(instance=medio)
144
145    return render(request, 'mpagos/form.html', {
146        'form': form,
147        'object': medio,
148        'cliente' : cliente,
149        'titulo': 'Editar Medio de Pago'
150    })

256@require_GET
257def stripe_registro_retorno(request):
258    """
259    Procesa el regreso desde Stripe luego de configurar
260    correctamente una tarjeta.
261    """
262
263    cliente = obtener_cliente_activo(request)
264
265    if not cliente:
266        messages.warning(
267            request,
268            "Debe seleccionar un cliente."
269        )
270        return redirect('dashboard')
271
272    if not puede_administrar_cliente(
273        request,
274        cliente
275    ):
276        return HttpResponseForbidden(
277            "No tiene permisos para registrar "
278            "tarjetas para este cliente."
279        )
280
281    setup_intent_id = request.GET.get(
282        'setup_intent'
283    )
284
285    if not setup_intent_id:
286        messages.error(
287            request,
288            "No se recibió la confirmación de Stripe."
289        )
290        return redirect('mpagos:crear')
291
292    try:
293        payment_method = obtener_tarjeta_confirmada(
294            cliente,
295            setup_intent_id
296        )
297
298        card = payment_method.card
299
300        predeterminado = (
301            request.GET.get('predeterminado')
302            == '1'
303        )
304
305        nombre_titular = (
306            payment_method.billing_details.name
307            or cliente.nombre_o_denominacion
308        )
309
310        with transaction.atomic():
311
312            existente = MedioPago.objects.filter(
313                stripe_payment_method_id=(
314                    payment_method.id
315                )
316            ).first()
317
318            if existente:
319
320                if existente.cliente_id != cliente.id:
321                    raise ValidationError(
322                        "La tarjeta ya se encuentra "
323                        "asociada a otro cliente."
324                    )
325
326                existente.activo = True
327                existente.nombre_titular = (
328                    nombre_titular
329                )
330                existente.numero_enmascarado = (
331                    f"**** **** **** {card.last4}"
332                )
333                existente.marca_tarjeta = (
334                    card.brand
335                )
336                existente.exp_mes = (
337                    card.exp_month
338                )
339                existente.exp_anio = (
340                    card.exp_year
341                )
342                existente.es_predeterminado = (
343                    predeterminado
344                )
345
346                medio = existente
347
348            else:
349
350                medio = MedioPago(
351                    usuario=request.user,
352                    cliente=cliente,
353                    tipo='TARJETA',
354                    nombre_titular=nombre_titular,
355                    banco_emisor=None,
356                    numero_enmascarado=(
357                        f"**** **** **** "
358                        f"{card.last4}"
359                    ),
360                    stripe_payment_method_id=(
361                        payment_method.id
362                    ),
363                    marca_tarjeta=card.brand,
364                    exp_mes=card.exp_month,
365                    exp_anio=card.exp_year,
366                    es_predeterminado=(
367                        predeterminado
368                    ),
369                    activo=True,
370                )
371
372            if predeterminado:
373                MedioPago.objects.filter(
374                    cliente=cliente,
375                    activo=True
376                ).exclude(
377                    stripe_payment_method_id=(
378                        payment_method.id
379                    )
380                ).update(
381                    es_predeterminado=False
382                )
383
384            medio.save()
385
386        messages.success(
387            request,
388            "Tarjeta registrada correctamente."
389        )
390
391        return redirect('mpagos:listar')
392
393    except ValidationError as error:
394
395        mensaje = (
396            error.message
397            if hasattr(error, 'message')
398            else str(error)
399        )
400
401        messages.error(
402            request,
403            mensaje
404        )
405
406    except Exception as error:
407
408        print(
409            "ERROR REGISTRO TARJETA STRIPE:",
410            type(error).__name__,
411            repr(error),
412            flush=True
413        )
414
415        messages.error(
416            request,
417            "No se pudo completar el registro "
418            "de la tarjeta."
419        )
420
421    return redirect('mpagos:crear')
